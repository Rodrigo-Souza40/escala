// ── LEITURA (GET) ───────────────────────────────────────── function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); if (!e || !e.parameter) { return out({values: ss.getSheets()[0].getDataRange().getDisplayValues()}); } var tabName = e.parameter.tab || ss.getSheets()[0].getName(); var s = ss.getSheetByName(tabName); if (!s) return out({error: 'Aba nao encontrada: ' + tabName}); return out({values: s.getDataRange().getDisplayValues()}); } // ── CRIAÇÃO DE ABA (POST) ────────────────────────────────── function doPost(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var payload = JSON.parse(e.postData.contents); var tabName = payload.tab; var srcTab = payload.srcTab; var matrix = payload.rows; if (!tabName || !matrix) return out({error: 'Faltam parametros'}); var s = ss.getSheetByName(tabName); if (s) { s.clearContents(); } else { var src = srcTab ? ss.getSheetByName(srcTab) : null; s = src ? src.copyTo(ss) : ss.insertSheet(tabName); s.setName(tabName); if (src) s.clearContents(); } if (matrix.length > 0) s.getRange(1, 1, matrix.length, matrix[0].length).setValues(matrix); if (srcTab) { var srcSheet = ss.getSheetByName(srcTab); if (srcSheet) { ss.setActiveSheet(s); ss.moveActiveSheet(srcSheet.getIndex() + 1); } } return out({ok: true, tab: tabName, rows: matrix.length}); } catch(err) { return out({error: err.message}); } } function out(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
