// Planilha usada pelo sistema (o Web App só acessa esta planilha) var SHEET_ID = '1o9VJIUjNP1XOMRnB2ZCRaiDOUAsbCFQGkjk4VsY3j9k'; var VERSION = 3; function doGet(e) { var p = (e && e.parameter) || {}; var ss = SpreadsheetApp.openById(SHEET_ID); if (p.action === 'tabs') { return out({ver: VERSION, tabs: ss.getSheets().map(function (s) { return s.getName(); })}); } if (p.action === 'plan') { var pl = readPlan_(ss, p.date); if (!pl) return out({ver: VERSION, error: 'Plano 5x1 não encontrado para ' + p.date + ' (importe a planilha 5x1 como aba)'}); return out({ver: VERSION, tab: pl.tab, date: p.date, members: pl.members.map(function (m) { return {g: m.g, role: m.role, key: m.key, short: m.short, st: m.days[p.date] || ''}; })}); } var s = p.tab ? ss.getSheetByName(p.tab) : ss.getSheets()[0]; if (!s) return out({error: 'Aba não encontrada: ' + p.tab}); var vals = s.getDataRange().getDisplayValues(); // notas da coluna Motorista ("Titular: ..."), alinhadas às linhas de vals var notes = [], h = findHeader_(vals); if (h && vals.length > h.hr + 1) { var nt = s.getRange(h.hr + 2, h.cm + 1, vals.length - h.hr - 1, 1).getNotes(); for (var i = 0; i < vals.length; i++) notes.push(i > h.hr ? nt[i - h.hr - 1][0] : ''); } return out({ver: VERSION, tab: s.getName(), values: vals, notes: notes}); } // Cria a nova aba copiando a aba modelo (mesma formatação), atualiza Dia e Data e aplica // os folguistas do plano 5x1. Nunca altera nem apaga uma aba existente. function doPost(e) { try { var p = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.openById(SHEET_ID); if (ss.getSheetByName(p.newTab)) return out({error: 'Já existe a aba "' + p.newTab + '". Nada foi alterado.'}); var src = ss.getSheetByName(p.srcTab); if (!src) return out({error: 'Aba modelo não encontrada: ' + p.srcTab}); var s = src.copyTo(ss); s.setName(p.newTab); var after = ss.getSheetByName(p.afterTab); ss.setActiveSheet(s); ss.moveActiveSheet(after ? after.getIndex() + 1 : ss.getNumSheets()); var last = s.getLastRow(), cols = s.getLastColumn(); var top = s.getRange(1, 1, Math.min(15, last), cols).getDisplayValues(); var h = findHeader_(top); if (!h) return out({ok: true, tab: p.newTab, rows: last, warn: ['Cabeçalho não encontrado: nada foi atualizado.']}); var hdr = top[h.hr].map(function (x) { return String(x).trim().toLowerCase(); }); var cd = hdr.indexOf('dia'), cdt = hdr.indexOf('data'), cm = h.cm; var co = hdr[cm + 1] === '' ? cm + 1 : -1; // coluna Obs (sem cabeçalho) logo após Motorista var n = last - (h.hr + 1), q = p.date.split('-'); var dt = new Date(+q[0], +q[1] - 1, +q[2]), txt = q[2] + '/' + q[1] + '/' + q[0]; var res = {ok: true, tab: p.newTab, rows: n, sub: 0, open: 0, warn: []}; if (n > 0) { setCol_(s, h.hr + 2, n, cd, function () { return p.dia; }); setCol_(s, h.hr + 2, n, cdt, function (old) { return old instanceof Date ? dt : txt; }); applyPlan_(ss, s, h.hr + 2, n, cm, co, p.date, res); } return out(res); } catch (err) { return out({error: err.message}); } } // Folguistas substituem quem está de folga no grupo. A célula guarda só quem trabalha; // a nota da célula registra o titular ("Titular: Fulano") para a próxima geração e para o dashboard. function applyPlan_(ss, s, row, n, cm, co, iso, res) { var plan = readPlan_(ss, iso); if (!plan) { res.warn.push('Plano 5x1 não encontrado para ' + iso + ' (importe a planilha 5x1 como aba). Nenhuma substituição aplicada.'); return; } var grp = {}; plan.members.forEach(function (m) { var g = grp[m.g] = grp[m.g] || {fix: [], fol: null}; if (m.role === 'folguista') g.fol = m; else g.fix.push(m); }); var rg = s.getRange(row, cm + 1, n, 1), vals = rg.getValues(), notes = rg.getNotes(); var orng = co >= 0 ? s.getRange(row, co + 1, n, 1) : null, obs = orng ? orng.getValues() : null; var seen = {}; for (var i = 0; i < n; i++) { var cell = String(vals[i][0]).trim(); var mt = /Titular:\s*([^\n(]+)/.exec(notes[i][0] || ''), T = cell, legacy = false; if (mt) T = mt[1].trim(); else if (cell.indexOf('/') > 0 && obs && /folga/i.test(String(obs[i][0]))) { T = cell.split('/')[0].trim(); legacy = true; } if (!T || T === 'SUBSTITUIR') continue; if (legacy && !match_(plan, T)) continue; // titular fora do plano: mantém a linha como está var r = resolve_(plan, grp, T, iso); vals[i][0] = r.name; if (r.titular) notes[i][0] = 'Titular: ' + r.titular + '\n' + (r.open ? 'Sem folguista disponível: linha aberta' : 'Folguista em campo (grupo ' + r.g + ')'); else if (mt) notes[i][0] = ''; if (obs) { if (r.open) obs[i][0] = 'ABERTA - ' + r.titular + ' em Folga'; else if (legacy || /^\s*(folga|aberta.*)$/i.test(String(obs[i][0]))) obs[i][0] = ''; } if (r.titular) { var k = r.titular + '|' + r.name; if (!seen[k]) { seen[k] = 1; if (r.open) res.open++; else res.sub++; } } } rg.setValues(vals); rg.setNotes(notes); if (orng) orng.setValues(obs); Object.keys(grp).forEach(function (g) { grp[g].fix.forEach(function (m) { if (!m.key && m.days[iso] === 'F') res.warn.push('Grupo ' + g + ': há Fixo sem nome no plano de folga em ' + iso + '; esse posto não foi substituído.'); }); }); } function resolve_(plan, grp, T, iso) { var m = match_(plan, T); if (!m || m.role !== 'fixo' || m.days[iso] !== 'F') return {name: T}; var f = grp[m.g] && grp[m.g].fol; if (f && f.short && f.days[iso] === 'T') return {name: f.short, titular: T, g: m.g}; return {name: 'SUBSTITUIR', titular: T, g: m.g, open: true}; } function match_(plan, name) { var k = nk_(name), hit = []; if (!k) return null; plan.members.forEach(function (m) { if (m.key && (m.key === k || m.key.indexOf(k + ' ') === 0 || k.indexOf(m.key + ' ') === 0)) hit.push(m); }); return hit.length === 1 ? hit[0] : null; } function nk_(s) { return String(s || '').toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '').replace(/[^a-z\s]/g, ' ').replace(/\s+/g, ' ').trim(); } // Plano 5x1: aba cujo nome contém "5x1" e cujas datas cobrem a data pedida function readPlan_(ss, iso) { var found = null; ss.getSheets().forEach(function (sh) { if (found || !/5\s*x\s*1/i.test(sh.getName())) return; var pl = parsePlan_(sh); if (pl.members.length && pl.dates.indexOf(iso) >= 0) found = pl; }); return found; } function parsePlan_(s) { var tz = s.getParent().getSpreadsheetTimeZone(), v = s.getDataRange().getValues(); var g = 0, cols = {}, members = [], dates = []; for (var i = 0; i < v.length; i++) { var r = v[i], role = String(r[1]).trim().toLowerCase(); if (r[0] !== '' && !isNaN(+r[0]) && /escala/i.test(String(r[1]))) g = +r[0]; var ds = {}, nd = 0; for (var j = 3; j < r.length; j++) if (r[j] instanceof Date) { ds[j] = Utilities.formatDate(r[j], tz, 'yyyy-MM-dd'); nd++; } if (nd >= 10) { cols = ds; continue; } if (role === 'fixo' || role === 'folguista') { var days = {}; for (var c in cols) { var t = String(r[c]).trim().toUpperCase(); if (t === 'T' || t === 'F') days[cols[c]] = t; } members.push({g: g, role: role, key: nk_(r[2]), days: days, short: String(r[2]).replace(/[^A-Za-zÀ-ÿ\s]/g, ' ').replace(/\s+/g, ' ').trim()}); } } for (var c2 in cols) dates.push(cols[c2]); return {tab: s.getName(), members: members, dates: dates}; } function findHeader_(v) { for (var i = 0; i < Math.min(15, v.length); i++) { for (var j = 0; j < v[i].length; j++) { if (String(v[i][j]).trim().toLowerCase() === 'motorista') return {hr: i, cm: j}; } } return null; } function setCol_(s, row, n, col, fn) { if (col < 0) return; var rg = s.getRange(row, col + 1, n, 1), v = rg.getValues(), f = rg.getFormulas(); for (var i = 0; i < n; i++) if (!f[i][0] && v[i][0] !== '') v[i][0] = fn(v[i][0]); rg.setValues(v); } function out(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
